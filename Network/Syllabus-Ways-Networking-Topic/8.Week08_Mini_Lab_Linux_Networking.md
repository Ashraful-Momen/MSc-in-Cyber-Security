## Day 1: Environment Setup (Create Everything)

```bash
#!/bin/bash
# lab_setup.sh — পুরো lab environment create

echo "=== Creating Lab Environment ==="

# 1. Cleanup any previous lab
sudo ip netns delete ns-a 2>/dev/null
sudo ip netns delete ns-b 2>/dev/null
sudo ip netns delete ns-c 2>/dev/null
sudo ip link delete br-lab 2>/dev/null

# 2. Create bridge
sudo ip link add name br-lab type bridge
sudo ip link set br-lab up
sudo ip addr add 10.100.0.254/24 dev br-lab
echo "✓ Bridge br-lab created (10.100.0.254/24)"

# 3. Create namespaces
sudo ip netns add ns-a
sudo ip netns add ns-b
sudo ip netns add ns-c
echo "✓ Namespaces ns-a, ns-b, ns-c created"

# 4. Create veth pairs
sudo ip link add veth-a type veth peer name veth-a-br
sudo ip link add veth-b type veth peer name veth-b-br
sudo ip link add veth-c type veth peer name veth-c-br
echo "✓ veth pairs created"

# 5. Move pod-end to namespaces
sudo ip link set veth-a netns ns-a
sudo ip link set veth-b netns ns-b
sudo ip link set veth-c netns ns-c
echo "✓ veth ends moved to namespaces"

# 6. Connect host-ends to bridge
sudo ip link set veth-a-br master br-lab
sudo ip link set veth-b-br master br-lab
sudo ip link set veth-c-br master br-lab
echo "✓ Host ends connected to bridge"

# 7. Configure namespace networking
# NS-A (Web Server)
sudo ip netns exec ns-a ip addr add 10.100.0.10/24 dev veth-a
sudo ip netns exec ns-a ip link set veth-a up
sudo ip netns exec ns-a ip link set lo up
sudo ip netns exec ns-a ip route add default via 10.100.0.254

# NS-B (Database Server)
sudo ip netns exec ns-b ip addr add 10.100.0.20/24 dev veth-b
sudo ip netns exec ns-b ip link set veth-b up
sudo ip netns exec ns-b ip link set lo up
sudo ip netns exec ns-b ip route add default via 10.100.0.254

# NS-C (Client)
sudo ip netns exec ns-c ip addr add 10.100.0.30/24 dev veth-c
sudo ip netns exec ns-c ip link set veth-c up
sudo ip netns exec ns-c ip link set lo up
sudo ip netns exec ns-c ip route add default via 10.100.0.254

# 8. Bring up host-end interfaces
sudo ip link set veth-a-br up
sudo ip link set veth-b-br up
sudo ip link set veth-c-br up

echo "✓ IP addresses and routes configured"

# 9. Enable NAT for internet access
sudo iptables -t nat -A POSTROUTING -s 10.100.0.0/24 ! -o br-lab -j MASQUERADE
sudo sysctl -w net.ipv4.ip_forward=1

echo "✓ NAT + IP Forwarding enabled"
echo ""
echo "=== Environment Ready ==="
echo "NS-A (Web):  10.100.0.10"
echo "NS-B (DB):   10.100.0.20"
echo "NS-C (Client): 10.100.0.30"
echo "Bridge:      10.100.0.254"
```

### Verify Setup

```bash
# Check bridge members
bridge link show br-lab

# Check ARP table
ip neigh show

# Check all namespaces
ip netns list

# Ping test from each namespace
sudo ip netns exec ns-a ping -c 2 10.100.0.20    # ns-a → ns-b
sudo ip netns exec ns-b ping -c 2 10.100.0.30    # ns-b → ns-c
sudo ip netns exec ns-c ping -c 2 10.100.0.10    # ns-c → ns-a
sudo ip netns exec ns-c ping -c 2 10.100.0.254   # ns-c → bridge

# Internet test
sudo ip netns exec ns-c ping -c 2 8.8.8.8
```

## Day 3: Analyze Captured Traffic

### Analysis with tcpdump

```bash
# Read the capture
tcpdump -n -r lab_traffic_bridge.pcap

# Filter HTTP only
tcpdump -n -r lab_traffic_bridge.pcap port 80

# Filter ICMP only
tcpdump -n -r lab_traffic_bridge.pcap icmp

# TCP SYN packets (connection initiation)
tcpdump -n -r lab_traffic_bridge.pcap 'tcp[tcpflags] & tcp-syn != 0'

# Count per protocol
tcpdump -n -r lab_traffic_bridge.pcap -q | awk '{print $1}' | sort | uniq -c
```

### Manual Analysis Questions

```bash
# Answer these by looking at the capture:

# 1. Three-way handshake-এ sequence numbers কী ছিল?
tcpdump -n -r lab_traffic_bridge.pcap 'tcp[tcpflags] & (tcp-syn|tcp-ack) != 0'

# 2. HTTP request-এ কোন headers ছিল?
tcpdump -A -r lab_traffic_bridge.pcap port 80 | head -50

# 3. ARP request/reply দেখো
tcpdump -n -r lab_traffic_bridge.pcap arp

# 4. Packet size distribution
tcpdump -n -r lab_traffic_bridge.pcap -v 2>&1 | grep -oP 'length \K\d+' | sort -n | uniq -c

# 5. Total data transferred
tcpdump -n -r lab_traffic_bridge.pcap -q | wc -l
```

## Day 4: Firewall Rules (iptables)

### Scenario 1: Allow only HTTP to Web Server

```bash
# Default deny on forward chain
sudo iptables -P FORWARD DROP

# Allow established/related (return traffic)
sudo iptables -A FORWARD -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT

# Allow NS-C → NS-A port 80 only
sudo iptables -A FORWARD -s 10.100.0.30 -d 10.100.0.10 -p tcp --dport 80 -j ACCEPT

# Test: HTTP should work
sudo ip netns exec ns-c curl http://10.100.0.10/
# ✓ Works

# Test: NS-C → NS-B should fail (no DB access allowed)
sudo ip netns exec ns-c nc -zv 10.100.0.20 5432
# ✗ Blocked (as expected)
```

### Scenario 2: Rate Limiting (DoS Protection)

```bash
# Limit ICMP to 1 per second from any source
sudo iptables -A FORWARD -p icmp --icmp-type echo-request \
  -m limit --limit 1/sec --limit-burst 3 -j ACCEPT
sudo iptables -A FORWARD -p icmp --icmp-type echo-request -j DROP

# Test: Fast ping → Rate limited
sudo ip netns exec ns-c ping -i 0.1 -c 10 10.100.0.10
# Should see some packets dropped
```

### Scenario 3: Port Forwarding (DNAT)

```bash
# External port 8080 → NS-A port 80
sudo iptables -t nat -A PREROUTING -p tcp --dport 8080 \
  -j DNAT --to-destination 10.100.0.10:80

# Test from host
curl http://localhost:8080/
# Should reach NS-A's HTTP server
```

### Check Rules

```bash
sudo iptables -L -n -v --line-numbers
sudo iptables -t nat -L -n -v
```

## Day 5: Network Troubleshooting Scenarios

### Scenario 1: "Can't reach web server"

```bash
# Diagnostic sequence (NS-C → NS-A):

# Step 1: Local interface
sudo ip netns exec ns-c ip addr show veth-c
# Is IP configured? Is interface UP?

# Step 2: Local route
sudo ip netns exec ns-c ip route show
# Is default route set?

# Step 3: Can I reach the bridge?
sudo ip netns exec ns-c ping 10.100.0.254

# Step 4: Can I reach the destination?
sudo ip netns exec ns-c ping 10.100.0.10

# Step 5: Is it an ARP problem?
sudo ip netns exec ns-c ip neigh show
# Does NS-A's IP appear with a MAC?

# Step 6: Is the service listening?
sudo ip netns exec ns-a ss -tlnp | grep 80
# Is port 80 in LISTEN state?

# Step 7: Is firewall blocking?
sudo iptables -L FORWARD -n -v
# Check counters on DROP rules

# Step 8: Capture to see what's happening
sudo tcpdump -i br-lab -n host 10.100.0.10
# Do packets reach the bridge? Does reply come back?
```

### Scenario 2: "Intermittent connectivity"

```bash
# Check interface errors
ip -s link show veth-a-br
# Look for: errors, dropped, overrun

# Check ARP table (STALE entries?)
ip neigh show
# STALE vs REACHABLE → timing issues

# Continuous ping with timestamp
sudo ip netns exec ns-c ping -D 10.100.0.10
# [-D = timestamp each line]
# Look for gaps, pattern of drops

# Check conntrack table (NAT issues?)
sudo conntrack -L | grep 10.100.0
```

## Day 6: Build Traffic Dataset (Research Practice)

### Collect Structured Traffic Data

```bash
#!/bin/bash
# traffic_collector.sh

PCAP_FILE="dataset_$(date +%Y%m%d_%H%M%S).pcap"
CSV_FILE="traffic_stats.csv"

# Header for CSV
echo "timestamp,src_ip,dst_ip,src_port,dst_port,protocol,packet_size,tcp_flags,ttl" > $CSV_FILE

# Capture for 5 minutes
sudo timeout 300 tcpdump -i br-lab -w $PCAP_FILE -v &

echo "Capturing for 5 minutes..."

# Generate varied traffic in background
for i in {1..50}; do
  sudo ip netns exec ns-c curl -s -o /dev/null http://10.100.0.10/ &
  sleep 1
  sudo ip netns exec ns-c ping -c 1 -W 0.5 10.100.0.20 &
  sleep 0.5
done

wait
echo "Capture complete: $PCAP_FILE"

# Extract features (requires tshark)
if command -v tshark &> /dev/null; then
  tshark -r $PCAP_FILE -T fields \
    -e frame.time_epoch \
    -e ip.src -e ip.dst \
    -e tcp.srcport -e tcp.dstport \
    -e ip.proto \
    -e frame.len \
    -e tcp.flags \
    -e ip.ttl \
    -E separator=, > $CSV_FILE
  echo "Features extracted: $CSV_FILE"
  echo "Total packets: $(wc -l < $CSV_FILE)"
else
  echo "Install tshark for feature extraction: sudo apt install tshark"
fi
```

### Analyze Dataset

```bash
# Quick stats
echo "=== Protocol Distribution ==="
awk -F',' 'NR>1 {print $5}' traffic_stats.csv | sort | uniq -c

echo "=== Top Src IPs ==="
awk -F',' 'NR>1 {print $2}' traffic_stats.csv | sort | uniq -c | sort -rn

echo "=== Top Dst Ports ==="
awk -F',' 'NR>1 {print $4}' traffic_stats.csv | sort -n | uniq -c | sort -rn

echo "=== Avg Packet Size ==="
awk -F',' 'NR>1 {sum+=$6; count++} END {print sum/count}' traffic_stats.csv
```

## Day 7: Integration + Documentation

### Final Integration Exercise

```bash
#!/bin/bash
# Complete scenario: Web App with DB Backend

# 1. NS-A runs web server (port 80)
sudo ip netns exec ns-a python3 -m http.server 80 &
WEB_PID=$!

# 2. NS-B runs DB server simulation
sudo ip netns exec ns-b nc -l -p 5432 -k &
DB_PID=$!

# 3. NS-C sends web requests every 5 seconds
for i in {1..20}; do
  echo "[$(date)] Request $i"
  sudo ip netns exec ns-c curl -s -o /dev/null -w "HTTP %{http_code}, Time: %{time_total}s\n" http://10.100.0.10/
  sleep 5
done

# 4. Meanwhile, monitor connections
watch -n 1 'ss -tan | grep 10.100.0'

# 5. Capture everything
sudo tcpdump -i br-lab -w final_lab.pcap

# Stop services
kill $WEB_PID $DB_PID 2>/dev/null
```
