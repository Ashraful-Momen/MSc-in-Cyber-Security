### Derived Features

```python
# Ratio features
df['byte_ratio'] = df['fwd_bytes'] / (df['fwd_bytes'] + df['bwd_bytes'] + 1)
df['packet_ratio'] = df['fwd_packets'] / (df['total_packets'] + 1)

# Rate features
df['packets_per_sec'] = df['total_packets'] / (df['duration'] + 1e-6)
df['bytes_per_sec'] = df['total_bytes'] / (df['duration'] + 1e-6)

# Z-score features (vs global mean)
df['bytes_zscore'] = (df['total_bytes'] - df['total_bytes'].mean()) / df['total_bytes'].std()

# Entropy features
def shannon_entropy(series):
    probs = series.value_counts(normalize=True)
    return -sum(probs * np.log2(probs))

df['port_entropy'] = df.groupby('src_ip')['dst_port'].transform(shannon_entropy)

# Time-based features
df['hour'] = pd.to_datetime(df['timestamp']).dt.hour
df['is_off_hours'] = ((df['hour'] < 7) | (df['hour'] > 19)).astype(int)
df['is_weekend'] = pd.to_datetime(df['timestamp']).dt.weekday.isin([5,6]).astype(int)

# Beacon detection (periodicity score)
def beacon_score(iat_series):
    """High score = regular intervals = beacon behavior"""
    if len(iat_series) < 3: return 0
    return 1.0 - iat_series.std() / (iat_series.mean() + 1e-6)

df['beacon_score'] = df.groupby('flow_id')['iat'].transform(beacon_score)

# Lateral movement score
df['is_internal_pair'] = (
    df['src_ip'].str.startswith(('10.', '172.16.', '192.168.')) &
    df['dst_ip'].str.startswith(('10.', '172.16.', '192.168.'))
)
df['new_destination'] = (~df['dst_ip'].isin(df.groupby('src_ip')['dst_ip'].transform(
    lambda x: x.expanding().apply(lambda y: y[:-1].tolist())
))).astype(int)
```

## 2. Normalization & Scaling

### Why Normalization Matters

```
ML models (especially distance-based) need normalized features:

Without normalization:
  bytes_sent = [100, 200, 500000, 300]  (range: 100 to 500K)
  port = [80, 443, 8080, 22]           (range: 22 to 8080)
  → bytes_sent dominates distance calculations!

With normalization:
  All features on comparable scale → each feature matters equally
```

### Methods

```python
from sklearn.preprocessing import StandardScaler, MinMaxScaler, RobustScaler

# StandardScaler: (x - mean) / std  → mean=0, std=1
# Best for: Normally distributed data (PCA, SVM, Logistic Reg)
scaler = StandardScaler()
df_scaled = scaler.fit_transform(df[numeric_cols])

# MinMaxScaler: (x - min) / (max - min) → range [0,1]
# Best for: Bounded data, Neural Networks
scaler = MinMaxScaler()
df_scaled = scaler.fit_transform(df[numeric_cols])

# RobustScaler: (x - median) / IQR
# Best for: Data with outliers! (Network data = lots of outliers)
scaler = RobustScaler()
df_scaled = scaler.fit_transform(df[numeric_cols])

# RECOMMENDATION for network security data:
# Use RobustScaler — handles outlier traffic spikes gracefully
```

## 3. Feature Selection Methods

### Filter Methods (Statistical)

```python
# 1. Variance Threshold: Remove low-variance features
from sklearn.feature_selection import VarianceThreshold
selector = VarianceThreshold(threshold=0.01)
selected = selector.fit_transform(df)

# 2. Correlation: Remove highly correlated features (redundant)
corr_matrix = df.corr().abs()
upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))
to_drop = [col for col in upper.columns if any(upper[col] > 0.95)]
df = df.drop(columns=to_drop)

# 3. Mutual Information: Best features vs target
from sklearn.feature_selection import mutual_info_classif
mi_scores = mutual_info_classif(df[features], df['label'])
top_features = pd.Series(mi_scores, index=features).nlargest(20)

# 4. ANOVA F-test: For classification
from sklearn.feature_selection import f_classif
f_scores, p_values = f_classif(df[features], df['label'])
```

### Wrapper Methods (Model-based)

```python
# RFE: Recursive Feature Elimination
from sklearn.feature_selection import RFE
from sklearn.ensemble import RandomForestClassifier

model = RandomForestClassifier()
```

## 4. Pipeline Implementation

```python
# Complete sklearn pipeline
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import RobustScaler, OneHotEncoder
from sklearn.feature_selection import SelectFromModel
from sklearn.ensemble import RandomForestClassifier

# Define feature groups
numeric_features = ['duration', 'bytes_sent', 'bytes_received',
                    'packets_sent', 'packets_received', 'packet_size_avg',
                    'iat_mean', 'iat_std', 'retransmit_count']
categorical_features = ['protocol', 'verdict', 'src_namespace', 'dst_namespace']
boolean_features = ['is_cross_namespace', 'is_off_hours', 'is_new_destination']

# Build preprocessor
preprocessor = ColumnTransformer([
    ('numeric', RobustScaler(), numeric_features),
    ('categorical', OneHotEncoder(drop='first', handle_unknown='ignore'), categorical_features),
    ('boolean', 'passthrough', boolean_features)
])

# Build full pipeline
pipeline = Pipeline([
    ('preprocessor', preprocessor),
    ('feature_selection', SelectFromModel(RandomForestClassifier(n_estimators=100))),
    ('classifier', RandomForestClassifier(n_estimators=200))
])

# Train
pipeline.fit(X_train, y_train)
y_pred = pipeline.predict(X_test)

# SHAP on pipeline (explain the classifier step)
explainer = shap.TreeExplainer(pipeline.named_steps['classifier'])
shap_values = explainer.shap_values(
    pipeline.named_steps['preprocessor'].transform(X_test)
)
```
