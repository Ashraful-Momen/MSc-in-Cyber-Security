## 🚀 Complete NumPy Master Cheat Sheet (for ML & DL)## ১. NumPy ইমপোর্ট এবং বেসিক অ্যারে তৈরি (Array Creation)

import numpy as np
# ১ডি (1D) এবং ২ডি (2D) অ্যারে তৈরিarr1d = np.array([1, 2, 3, 4])arr2d = np.array([[1, 2, 3], [4, 5, 6]])
# নির্দিষ্ট রেঞ্জের অ্যারে তৈরিarr_range = np.arange(0, 10, 2)       # Output: [0, 2, 4, 6, 8]arr_linear = np.linspace(0, 1, 5)     # ০ থেকে ১ এর মধ্যে সমান দূরত্বের ৫টি সংখ্যা -> [0., 0.25, 0.5, 0.75, 1.]
# বিশেষ অ্যারে (ML Weights & Biases ইনিশিয়ালাইজেশনে লাগে)zeros = np.zeros((2, 3))              # ২ রো, ৩ কলামের সব ০.০ones = np.ones((3, 4))                # ৩ রো, ৪ কলামের সব ১.০identity = np.eye(3)                  # ৩x৩ আইডেন্টিটি ম্যাট্রিক্স (ডায়াগোনাল বরাবর ১)

## ২. র্যান্ডম ডেটা জেনারেশন (Random Sampling - ML-এ প্রচুর লাগে)

# ০ থেকে ১ এর মধ্যে র্যান্ডম ভ্যালু তৈরিrand_uniform = np.random.rand(2, 3)   # ২x৩ ম্যাট্রিক্স (Uniform Distribution)
# Standard Normal Distribution (Mean = 0, Std Dev = 1) থেকে ডেটা নেওয়াrand_normal = np.random.randn(3, 3)   # ৩x৩ ম্যাট্রিক্স (Gaussian Distribution)
# নির্দিষ্ট রেঞ্জের মধ্যে ইন্টিজার (পূর্ণসংখ্যা) তৈরিrand_int = np.random.randint(1, 100, size=(5,)) # ১ থেকে ৯৯ এর মধ্যে ৫টি র্যান্ডম সংখ্যা

## ৩. অ্যারের বৈশিষ্ট্য চেক করা (Array Attributes)

# ধরুন, b = np.array([[1, 2, 3], [4, 5, 6]])
print(b.shape)  # Output: (2, 3) -> (রো, কলাম) সংখ্যা
print(b.ndim)   # Output: 2      -> ডায়মেনশন বা মাত্রা (1D, 2D নাকি 3D)
print(b.dtype)  # Output: int64  -> ডেটা টাইপ (int64, float64 ইত্যাদি)
print(b.size)   # Output: 6      -> অ্যারেতে মোট উপাদানের সংখ্যা (২ x ৩ = ৬)

## ৪. ইনডেক্সিং এবং স্লাইসিং (Indexing & Slicing - ডেটা স্প্লিট করার জন্য)
ML-এ ফিচার (X) এবং লেবেল (y) আলাদা করতে এটি সবচেয়ে গুরুত্বপূর্ণ।

# Syntax: array[row_start:row_end, col_start:col_end]data = np.array([,
 ,
    [70, 80, 90, 1]
])
# সব রো এবং প্রথম ৩টি কলাম (Features - X) নেওয়াX = data[:, 0:3]  # Output: [[10, 20, 30], [40, 50, 60], [70, 80, 90]]
# সব রো এবং একদম শেষ কলামটি (Target/Label - y) নেওয়াy = data[:, -1]   # Output: [1, 0, 1]
# নির্দিষ্ট কোনো উপাদান অ্যাক্সেস করা (যেমন: ২য় রো, ৩য় কলাম)element = data[1, 2] # Output: 60

## ৫. শেপ পরিবর্তন (Reshaping & Transforming)

# Reshape: উপাদান সংখ্যা ঠিক রেখে ম্যাট্রিক্সের ডাইমেনশন বদলানোc = np.array([1, 2, 3, 4, 5, 6])matrix = c.reshape(2, 3)       # ২x৩ ম্যাট্রিক্সে রূপান্তর column_vector = c.reshape(-1, 1) # অটোমেটিক রো হিসাব করে ৬x১ কলাম ভেক্টর বানাবে (ML-এ খুব কাজে দেয়)
# Flatten: যেকোনো মাল্টি-ডায়মেনশনাল অ্যারেকে ১ডি (1D) বানানোflat_arr = matrix.flatten()    # Output: [1, 2, 3, 4, 5, 6]
# Transpose: রো-কে কলাম আর কলাম-কে রো বানানো (Matrix Multiplication এর জন্য)transposed_matrix = matrix.T   # ৩x২ ম্যাট্রিক্সে পরিণত হবে

## ৬. গাণিতিক অপারেশন ও ম্যাট্রিক্স গুণ (Math & Matrix Multiplication)

x = np.array([1, 2, 3])y = np.array([4, 5, 6])
# Element-wise Operations (সাধারণ গাণিতিক কাজ)
print(x + y)  # Output: [5, 7, 9]
print(x * 2)  # Output: [2, 4, 6]
print(x ** 2) # Output: [1, 4, 9]
# Matrix Multiplication / Dot Product (ML/DL এর মূল হৃদপিণ্ড)A = np.array([[1, 2], [3, 4]])B = np.array([[5, 6], [7, 8]])
output1 = A @ B          # আধুনিক পাইথন স্টাইল (Recommended)output2 = np.dot(A, B)   # ট্র্যাডিশনাল NumPy স্টাইল# Both Output: [[19, 22], [43, 50]]

## ৭. পরিসংখ্যান ও অক্ষভিত্তিক হিসাব (Aggregations & Axis)

# b = np.array([[1, 2], [3, 4]])
print(b.sum())   # সব উপাদানের যোগফল -> 10
print(b.mean())  # গড় (Average) -> 2.5
print(b.std())   # স্ট্যান্ডার্ড ডেভিয়েশন (Standard Deviation) -> 1.118...
# ---- Axis বরাবর হিসাব (ML Loss/Accuracy ক্যালকুলেশনে ব্যবহৃত হয়) ----# axis=0 মানে কলাম বরাবর (Down)
print(b.sum(axis=0))  # Output: [4, 6]
# axis=1 মানে রো বরাবর (Across)
print(b.mean(axis=1)) # Output: [1.5, 3.5]

## ৮. সর্টিং এবং সার্চিং (Sorting & Searching)

arr = np.array([30, 10, 20])
# ছোট থেকে বড় সাজানোsorted_arr = np.sort(arr)   # Output: [10, 20, 30]
# সর্ট করলে ইনডেক্সগুলোর পজিশন কেমন হতো তা জানা (ML-এ কনফিডেন্স স্কোর ইনডেক্স বের করতে লাগে)idx = np.argsort(arr)       # Output: [1, 2, 0]
# সর্বোচ্চ এবং সর্বনিম্ন মানের ইনডেক্স বের করাmax_idx = np.argmax(arr)    # Output: 0 (কারণ ৩০ সবার বড় এবং এর ইনডেক্স ০)min_idx = np.argmin(arr)    # Output: 1 (১০ সবার ছোট এবং এর ইনডেক্স ১)

## ৯. ফিল্টারিং এবং কন্ডিশন (Boolean Masking)
ডেটা ক্লিন করার সময় বা আউটলায়ার বাদ দিতে এটি ব্যবহার করা হয়।

ages = np.array([15, 25, 42, 18, 30])
# কন্ডিশন দিয়ে একটি মাস্ক তৈরি করাmask = ages >= 21  # Output: [False,  True,  True, False,  True]
# মাস্ক ফিল্টার হিসেবে ব্যবহার করে ডেটা নেওয়াadults = ages[mask] # Output: [25, 42, 30]
# np.where: কন্ডিশনের ওপর ভিত্তি করে ভ্যালু অ্যাসাইন করা (যেমন: Binary Classification)# Syntax: np.where(condition, if_true, if_false)labels = np.where(ages >= 18, "Adult", "Minor")# Output: ['Minor', 'Adult', 'Adult', 'Adult', 'Adult']

## 💡 প্রো-টিপ (Pro-Tip for Beginners):
আপনি যখন পরবর্তীতে PyTorch বা TensorFlow শিখবেন, তখন এই ফাংশনগুলোর নাম ও কাজ প্রায় ৯৯% একই থাকবে। যেমন:

* NumPy-এর arr.reshape() হয়ে যাবে PyTorch-এর tensor.view() বা tensor.reshape()।
* NumPy-এর arr.argmax() হয়ে যাবে PyTorch-এর tensor.argmax()।
* NumPy-এর axis=0 হয়ে যাবে PyTorch-এর dim=0।
