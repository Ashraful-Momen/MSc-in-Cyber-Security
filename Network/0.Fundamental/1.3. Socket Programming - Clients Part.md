# client.py

```python
"""
======================================================
        Simple TCP Chat Client
======================================================

Features

✔ Username

✔ Send Message

✔ Receive Message

✔ Multi Client

"""

# ====================================================
# Socket Library
# ====================================================
import socket

# ====================================================
# Thread Library
# ====================================================
import threading

# ====================================================
# Server Address
# ====================================================
HOST = "127.0.0.1"
PORT = 5000

# ====================================================
# Create TCP Socket
#
# AF_INET
# IPv4
#
# SOCK_STREAM
# TCP
# ====================================================
client = socket.socket(
    socket.AF_INET,
    socket.SOCK_STREAM
)

print("=" * 50)
print(" TCP Chat Client ")
print("=" * 50)

# ====================================================
# Username Input
#
# Example
#
# Alice
# Bob
# Charlie
# ====================================================
username = input("Enter Username : ")

# ====================================================
# Connect Server
#
# Internally
#
# SYN
#
# SYN + ACK
#
# ACK
#
# Happens Automatically
# ====================================================
client.connect((HOST, PORT))

print("Connected Successfully\n")

# ====================================================
# First Message
#
# Send Username
#
# Server will store it
# ====================================================
client.send(username.encode())


# ====================================================
# Receive Messages
#
# This function runs forever
#
# in another Thread
# ====================================================
def receive_message():

    while True:

        try:

            # Receive Broadcast Message
            message = client.recv(1024)

            print("\n" + message.decode())

        except:

            print("Disconnected From Server")

            client.close()

            break


# ====================================================
# Send Messages
#
# Infinite Loop
# ====================================================
def send_message():

    while True:

        # User Input
        message = input()

        # Ignore Empty Message
        if message.strip() == "":
            continue

        # Send To Server
        client.send(message.encode())


# ====================================================
# Receive Thread
#
# Server Message শুনবে
# ====================================================
receive_thread = threading.Thread(
    target=receive_message
)

receive_thread.start()


# ====================================================
# Send Thread
#
# Keyboard Input নিবে
# ====================================================
send_thread = threading.Thread(
    target=send_message
)

send_thread.start()
```
