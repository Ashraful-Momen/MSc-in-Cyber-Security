## Day 4: Protocol Analysis — Attack Traffic

### Compare Normal vs Attack

```bash
# Create analysis function
analyze_pcap() {
    PCAP=$1
    LABEL=$2
    
    echo "========================================"
    echo "  ANALYSIS: $LABEL"
    echo "========================================"
    
    # Packet count
    PKT_COUNT=$(tshark -r $PCAP | wc -l)
    echo "Total Packets: $PKT_COUNT"
    
    # TCP SYN count (port scan indicator)
    SYN_COUNT=$(tshark -r $PCAP -Y "tcp.flags.syn==1 && tcp.flags.ack==0" | wc -l)
    echo "Pure SYN Packets: $SYN_COUNT"
    
    # RST count (closed ports)
    RST_COUNT=$(tshark -r $PCAP -Y "tcp.flags.reset==1" | wc -l)
    echo "RST Packets: $RST_COUNT"
    
    # SYN/RST ratio
    if [ "$RST_COUNT" -gt 0 ]; then
        echo "SYN/RST Ratio: $(echo "scale=2; $SYN_COUNT/$RST_COUNT" | bc)"
    fi
    
    # Unique destination IPs
    UNIQUE_IP=$(tshark -r $PCAP -T fields -e ip.dst | sort -u | wc -l)
    echo "Unique Destination IPs: $UNIQUE_IP"
    
    # Unique destination ports
    UNIQUE_PORT=$(tshark -r $PCAP -T fields -e tcp.dstport | sort -u | wc -l)
    echo "Unique Destination Ports: $UNIQUE_PORT"
    
    # Retransmission count
    RETX=$(tshark -r $PCAP -Y "tcp.analysis.retransmission" | wc -l)
    echo "Retransmissions: $RETX"
    
    # DNS NXDOMAIN
    NX=$(tshark -r $PCAP -Y "dns.flags.rcode == 3" | wc -l)
    echo "DNS NXDOMAIN responses: $NX"
    
    # HTTP Error rate
    HTTP_TOTAL=$(tshark -r $PCAP -Y "http.response" | wc -l)
    HTTP_4XX=$(tshark -r $PCAP -Y "http.response.code >= 400 && http.response.code < 500" | wc -l)
    HTTP_5XX=$(tshark -r $PCAP -Y "http.response.code >= 500" | wc -l)
    echo "HTTP 4xx Errors: $HTTP_4XX / $HTTP_TOTAL"
    echo "HTTP 5xx Errors: $HTTP_5XX / $HTTP_TOTAL"
    
    echo ""
}

# Run for each capture
analyze_pcap normal_browsing.pcap "NORMAL BROWSING"
analyze_pcap normal_api.pcap "NORMAL API"
analyze_pcap attack_portscan.pcap "ATTACK: Port Scan"
analyze_pcap attack_http_suspicious.pcap "ATTACK: HTTP Suspicious"
analyze_pcap attack_dns_suspicious.pcap "ATTACK: DNS Suspicious"
```

## Day 5-6: Feature Extraction + Dataset Creation

### Pipeline Script

```bash
#!/bin/bash
# build_dataset.sh — Complete dataset building pipeline

DATASET_DIR="../dataset"
mkdir -p "$DATASET_DIR"

MASTER_CSV="$DATASET_DIR/master_dataset.csv"

# Write header (once)
echo "timestamp,src_ip,dst_ip,src_port,dst_port,protocol,packet_size,tcp_flags,ttl,label" > "$MASTER_CSV"

# Process each capture
process_pcap() {
    PCAP="$1"
    LABEL="$2"
    
    echo "[*] Processing $PCAP (label: $LABEL)..."
    
    tshark -r "$PCAP" -T fields \
        -e frame.time_epoch \
        -e ip.src \
        -e ip.dst \
        -e tcp.srcport \
        -e tcp.dstport \
        -e ip.proto \
        -e frame.len \
        -e tcp.flags \
        -e ip.ttl \
        -E separator=, \
        -E occurrence=f \
        2>/dev/null | while IFS= read -r line; do
        [ -z "$line" ] && continue
        echo "$line,$LABEL"
    done >> "$MASTER_CSV"
}

# Process all files
process_pcap "normal_browsing.pcap" "normal"
process_pcap "normal_api.pcap" "normal"
process_pcap "attack_portscan.pcap" "attack"
process_pcap "attack_synflood.pcap" "attack"
process_pcap "attack_dns_suspicious.pcap" "attack"
process_pcap "attack_http_suspicious.pcap" "attack"

TOTAL=$(wc -l < "$MASTER_CSV")
NORMAL=$(grep -c "normal$" "$MASTER_CSV")
ATTACK=$(grep -c "attack$" "$MASTER_CSV")

echo ""
echo "=== Dataset Complete ==="
echo "Total rows:  $TOTAL"
echo "Normal:      $NORMAL (normal)"
echo "Attack:      $ATTACK (attack)"
echo "File:        $MASTER_CSV"
```

### Python Analysis + Validation

```python
#!/usr/bin/env python3
"""dataset_analysis.py — Validate and analyze the traffic dataset"""

import pandas as pd
import numpy as np

# Load dataset
df = pd.read_csv('../dataset/master_dataset.csv')

print("=" * 60)
print("DATASET ANALYSIS REPORT")
print("=" * 60)

# Basic stats
print(f"\nTotal Samples: {len(df)}")
print(f"Features: {len(df.columns)}")
print(f"\nClass Distribution:")
print(df['label'].value_counts())
print(f"\nImbalance Ratio: {df['label'].value_counts()['normal'] / df['label'].value_counts()['attack']:.2f}")

# Missing values
print(f"\nMissing Values:")
print(df.isnull().sum())

# Numeric feature stats
numeric_cols = ['packet_size', 'ttl']
print(f"\nNumeric Feature Statistics:")
print(df[numeric_cols].describe())

# Protocol distribution per class
print(f"\nProtocol Distribution by Class:")
print(pd.crosstab(df['label'], df['protocol']))

# TCP flags distribution (attack vs normal)
if 'tcp_flags' in df.columns:
    print(f"\nTCP Flag Patterns:")
    # Common flag values in hex
    for label in ['normal', 'attack']:
        subset = df[df['label'] == label]['tcp_flags'].value_counts().head(5)
        print(f"  {label}:")
        for val, cnt in subset.items():
            # Decode common TCP flags
            flag_bits = int(str(val), 16) if pd.notna(val) else 0
            flags = []
            if flag_bits & 0x02: flags.append('SYN')
            if flag_bits & 0x10: flags.append('ACK')
            if flag_bits & 0x01: flags.append('FIN')
            if flag_bits & 0x04: flags.append('RST')
            if flag_bits & 0x08: flags.append('PSH')
            flag_str = '+'.join(flags) if flags else str(val)
            print(f"    0x{val:04x} ({flag_str}): {cnt}")

# Save cleaned dataset
df_clean = df.dropna(subset=['src_ip', 'dst_ip'])
df_clean.to_csv('../dataset/dataset_clean.csv', index=False)
print(f"\nCleaned dataset saved: {len(df_clean)} rows")
```
