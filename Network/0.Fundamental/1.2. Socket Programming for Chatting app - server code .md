# Server Source Code

```python
"""
=========================================================
        Simple TCP Multi Client Chat Server
=========================================================

Features

✔ Multiple Client
✔ Username
✔ Broadcast
✔ Thread
✔ Simple Code

"""

# =====================================================
# Import socket library
# =====================================================
import socket

# =====================================================
# Thread ব্যবহার করার জন্য threading import করছি
# =====================================================
import threading


# =====================================================
# Server Configuration
# =====================================================

# localhost
HOST = "127.0.0.1"

# Listening Port
PORT = 5000


# =====================================================
# Connected Client Socket রাখবে
# Example
#
# clients =
# [
#     socket1,
#     socket2,
#     socket3
# ]
# =====================================================
clients = []


# =====================================================
# Username রাখবে
#
# usernames =
#
# {
#     socket1 : "Alice",
#     socket2 : "Bob",
#     socket3 : "Charlie"
# }
# =====================================================
usernames = {}


# =====================================================
# Create TCP Socket
#
# AF_INET
#     IPv4
#
# SOCK_STREAM
#     TCP
# =====================================================
server = socket.socket(
    socket.AF_INET,
    socket.SOCK_STREAM
)


# =====================================================
# Bind Server
#
# Server এখন
#
# 127.0.0.1:5000
#
# Address এ Listen করবে
# =====================================================
server.bind((HOST, PORT))


# =====================================================
# Start Listening
#
# Maximum 5 pending connection queue
# =====================================================
server.listen(5)


print("=" * 50)
print(" TCP Chat Server Started ")
print("=" * 50)
print(f"Listening : {HOST}:{PORT}")
print()


# =====================================================
# Broadcast Function
#
# একজন Client Message পাঠালে
#
# অন্য সবাইকে Send করবে
# =====================================================
def broadcast(message, sender):

    # সকল connected client এর উপর loop
    for client in clients:

        # যে send করেছে
        # তাকে আর send করবো না
        if client != sender:

            try:

                client.send(message)

            except:

                # যদি client disconnected হয়

                if client in clients:
                    clients.remove(client)

                if client in usernames:
                    del usernames[client]


# =====================================================
# Handle Each Client
#
# প্রত্যেক Client
#
# আলাদা Thread এ চলবে
# =====================================================
def handle_client(client):

    while True:

        try:

            # Client থেকে Message Receive
            message = client.recv(1024)

            # যদি কিছু receive হয়
            if message:

                # Username বের করা
                username = usernames[client]

                # Final Message
                final_message = (
                    f"{username}: "
                    + message.decode()
                )

                print(final_message)

                # সবাইকে Send
                broadcast(
                    final_message.encode(),
                    client
                )

            else:

                break

        except:

            break

    # ===========================
    # Client Disconnect
    # ===========================

    print(
        usernames[client],
        "Disconnected"
    )

    clients.remove(client)

    del usernames[client]

    client.close()


# =====================================================
# Main Server Loop
#
# Server সবসময়
#
# নতুন Client এর জন্য অপেক্ষা করবে
# =====================================================
while True:

    print("Waiting for New Client...")

    # ==========================================
    # এখানে TCP 3-Way Handshake Complete হয়।
    #
    # Client connect() কল করলে OS Kernel
    # SYN -> SYN+ACK -> ACK সম্পন্ন করে।
    #
    # Handshake সফল হলেই accept() return করে।
    # ==========================================
    client, address = server.accept()

    print(f"Connected : {address}")

    # ==========================================
    # Username Receive
    #
    # Client connect হওয়ার পর
    # প্রথম Message হিসেবে Username পাঠাবে।
    # ==========================================
    username = client.recv(1024).decode()

    print(username, "Joined")

    # ==========================================
    # Save Client
    # ==========================================
    clients.append(client)

    usernames[client] = username

    # ==========================================
    # সবাইকে Notify
    # ==========================================
    broadcast(
        f"{username} Joined the Chat".encode(),
        client
    )

    # ==========================================
    # Create Thread
    #
    # এই Client এখন
    #
    # নিজের Thread এ চলবে।
    # ==========================================
    thread = threading.Thread(
        target=handle_client,
        args=(client,)
    )

    thread.start()
```
